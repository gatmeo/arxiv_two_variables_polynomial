\section{Proof of the main theorem}
\label{chapter_main_thm}

Let
\[
  X=X^!=\widetilde{\nilcone}\longrightarrow\nilcone
\]
be the self-dual type $A$ Springer resolution. Set
\[
  l(\lambda):=\binom n2-n(\lambda)
  =\tfrac12\dim\mathcal O_\lambda.
\]
We prove the following more precise form of
\Cref{thm:bichar_full_flag}.

\begin{lemma}[lem:main_product_filtration]{Main lemma}
  The contribution indexed by $\lambda\vdash n$ to the
  bicharacteristic polynomial is
  \[
    x^{l(\lambda)}Q(\mathcal O_\lambda,x^{-1})
    y^{l(\lambda^T)}Q(\mathcal O_{\lambda^T},y^{-1})
    =K_{\lambda,(1^n)}(x)K_{\lambda^T,(1^n)}(y).
  \]
  Consequently,
  \[
    T_{X,X^!}(x,y)
    =\sum_{\lambda\vdash n}
      K_{\lambda,(1^n)}(x)K_{\lambda^T,(1^n)}(y).
  \]
\end{lemma}

\subsection{The Springer support filtration}

Let
\[
  \misom:\HH_T(X)_\sigma\xrightarrow{\sim}\HH_T(X)_\eta
\]
be the stable-envelope isomorphism of \Cref{eqn:m}. The dominant
integral parameters $(\sigma,\eta)$ determine Koszul-dual categories
$\mathcal O$ and $\mathcal O^!$ and identifications
\[
  K(\mathcal O)\cong\HH_T(X)_\sigma,
  \qquad
  K(\mathcal O^!)\cong\HH_T(X)_\eta.
\]
Write $P_w,L_w,S_w$ for the projective, simple, and standard classes in
$K(\mathcal O)$, and use a superscript $!$ for the corresponding
classes in $K(\mathcal O^!)$. Each of these three families is a basis,
and the standard basis agrees with the stable-envelope basis.

For $w\in W$, let $\lambda_w$ be determined by
\[
  G\cdot(\lie n\cap w\cdot\lie n)=\mathcal O_{\lambda_w}.
\]
The category-$\mathcal O$ description of the support filtration gives
the following statement.

\begin{proposition}[prop:bdd_spans]{}
  Define
  \[
    F_{\leq\lambda}
      :=\operatorname{span}\{[P_w]\mid\lambda_w\leq\lambda\},
    \qquad
    F_{\geq\lambda}
      :=\operatorname{span}\{[L_w]\mid\lambda_w\geq\lambda\}.
  \]
  Under the decomposition of \Cref{crl:type_A_decomposition}, these are
  respectively the sums of the terms supported on
  $\overline{\mathcal O_\mu}$ with $\mu\leq\lambda$ and with
  $\mu\geq\lambda$.
\end{proposition}

Choose local polarizations
\[
  \varepsilon_w=(-1)^{\ell(w)},
  \qquad \varepsilon^!_{w^!}=1,
  \qquad w^!=w^{-1}w_\circ.
\]
By \cite[Remark 10.28]{braden2016quantizationsconicalsymplecticresolutions},
\[
  \misom([S_w])=(-1)^{\ell(w)}[S^!_{w^!}],
  \quad
  \misom([L_w])=(-1)^{\ell(w)}[P^!_{w^!}],
  \quad
  \misom([P_w])=(-1)^{\ell(w)}[L^!_{w^!}].
\]
The Robinson--Schensted description in \Cref{thm:irr_steinberg}
implies
\[
  \lambda_{w^!}=\lambda_w^T.
\]
Since transposition reverses dominance order, it follows that
\[
  \misom(F_{\leq\lambda})=F^!_{\geq\lambda^T}.
\]

For a partition $\lambda$, set
\[
  \operatorname{gr}_\lambda F
  :=F_{\leq\lambda}\Big/
    \sum_{\mu<\lambda}F_{\leq\mu},
  \qquad
  \operatorname{gr}_{\lambda^T}F^!
  :=F^!_{\geq\lambda^T}\Big/
    \sum_{\nu>\lambda^T}F^!_{\geq\nu}.
\]
Thus $\misom$ induces an isomorphism
\begin{equation}
  \label{eq:m_on_support_graded}
  \operatorname{gr}_\lambda(\misom):
  \operatorname{gr}_\lambda F
  \xrightarrow{\sim}
  \operatorname{gr}_{\lambda^T}F^!.
\end{equation}
After identifying these subquotients through the semismall
decomposition, this becomes
\[
  \IH^\bullet(\overline{\mathcal O_\lambda})
    \otimes\Htop(\BB_\lambda)
  \xrightarrow{\sim}
  \IH^\bullet(\overline{\mathcal O_{\lambda^T}})
    \otimes\Htop(\BB_{\lambda^T}).
\]
This is an assertion about the associated graded of the support
filtration; it does not require a canonical splitting of the
decomposition theorem.

\begin{proposition}[prop:m_preserves_summand]{}
  The stable-envelope isomorphism induces the isomorphism
  \eqref{eq:m_on_support_graded} for every $\lambda\vdash n$.
\end{proposition}

\subsection{A linear-algebra lemma}

We write all unadorned tensor products and Hom spaces over $\mathbb C$.

\begin{lemma}[lem:tensor_hom]{}
  Let $M,M'$ be finite-dimensional $A$-modules and let $N,N'$ be
  finite-dimensional $B$-modules. Then the natural map
  \[
    \operatorname{Hom}_A(M,M')\otimes\operatorname{Hom}_B(N,N')
    \longrightarrow
    \operatorname{Hom}_{A\otimes B}(M\otimes N,M'\otimes N')
  \]
  is an isomorphism.
\end{lemma}

\begin{lemma}[lem:commutes_implies_split]{}
  Let $M$ be a simple $A$-module and $N$ a simple $B$-module. Suppose
  \[
    f:V\otimes M\xrightarrow{\sim}W\otimes N
  \]
  transports the $B$-action to one commuting with the original
  $A$-action. If $\dim M=\dim W$ and $\dim V=\dim N$, then
  $f=f_1\otimes f_2$ for vector-space isomorphisms
  $f_1:V\to N$ and $f_2:M\to W$.
  \begin{proof}
    By \Cref{lem:tensor_hom} and Schur's lemma,
    \[
      \operatorname{End}_A(V\otimes M)
      \cong\operatorname{End}(V)\otimes\operatorname{End}_A(M)
      \cong\operatorname{End}(V).
    \]
    Hence the transported $B$-action comes from $V$; similarly, the
    transported $A$-action comes from $W$. Therefore
    \[
      f\in
      \operatorname{Hom}_A(M,W)\otimes\operatorname{Hom}_B(V,N).
    \]
    Simplicity, the dimension assumptions, and invertibility of $f$
    imply that both Hom spaces are one-dimensional and generated by
    isomorphisms. The result follows.
  \end{proof}
\end{lemma}

\subsection{The two Weyl-group actions}

Transporting the left $\HBMtop(\steinberg)$-action through $\misom$
gives a second action on $\HH_T(X)_\eta$.

\begin{proposition}[prop:action_commute]{}
  The transported and original $\HBMtop(\steinberg)\cong\mathbb C[W]$
  actions on $\HH_T(X)_\eta$ commute.
  \begin{proof}
    Embed both equivariant cohomology modules by localization into
    $\mathbb C[\lie h\times\lie h]^{\oplus W}$, and write $1_{w'}$ for
    the fixed-point basis. By \Cref{lem:W_action_on_stab}, the first
    action is
    \[
      w*1_{w'}=1_{w'w^{-1}}.
    \]
    Using $w^!=w^{-1}w_\circ$ to express the fixed-point identification
    on the dual side, the second action is
    \[
      w*^!1_{w'}=1_{w_\circ w w_\circ w'}.
    \]
    These are respectively right and left multiplication on the
    indexing set $W$, and therefore commute.
  \end{proof}
\end{proposition}

\subsection{Completion of the proof}

\begin{proof}[Proof of \Cref{lem:main_product_filtration}]
  On the $\lambda$-subquotient, \eqref{eq:m_on_support_graded} has the
  form
  \[
    \IH^\bullet(\overline{\mathcal O_\lambda})
      \otimes\Htop(\BB_\lambda)
    \longrightarrow
    \IH^\bullet(\overline{\mathcal O_{\lambda^T}})
      \otimes\Htop(\BB_{\lambda^T}).
  \]
  The Weyl group acts only on the Springer-representation factors
  $\Htop(\BB_\lambda)$ and $\Htop(\BB_{\lambda^T})$, which are simple
  by \Cref{thm:springer_correspondence}. Moreover,
  \[
    \dim\Htop(\BB_\lambda)
      =|\mathcal Y_\lambda|
      =|\mathcal Y_{\lambda^T}|
      =\dim\IH^\bullet(\overline{\mathcal O_{\lambda^T}}),
  \]
  where we use \Cref{thm:irred_springer} and \Cref{thm:IH_kostka}; the analogous
  equality holds after interchanging $\lambda$ and $\lambda^T$.
  By \Cref{prop:action_commute} and \Cref{lem:commutes_implies_split}, the induced
  map exchanges the two tensor factors, up to vector-space
  isomorphisms.

  It follows that the two cohomological filtrations on this subquotient
  are the product of the gradings on
  $\IH^\bullet(\overline{\mathcal O_\lambda})$ and
  $\IH^\bullet(\overline{\mathcal O_{\lambda^T}})$. Its contribution
  to the bicharacteristic polynomial is therefore
  \[
    x^{l(\lambda)}Q(\mathcal O_\lambda,x^{-1})
    y^{l(\lambda^T)}Q(\mathcal O_{\lambda^T},y^{-1}).
  \]
  Applying \Cref{thm:IH_kostka} converts this expression to
  \[
    K_{\lambda,(1^n)}(x)K_{\lambda^T,(1^n)}(y).
  \]
  Summing over $\lambda\vdash n$ proves the lemma and
  \Cref{thm:bichar_full_flag}.
\end{proof}

\begin{example}[exp:bichar_n3]{}
  Let $n=3$. The partitions are $(3)$, $(2,1)$, and $(1,1,1)$, and the
  charge formula gives
  \[
    K_{(3),(1^3)}(q)=q^3,
    \qquad
    K_{(2,1),(1^3)}(q)=q+q^2,
    \qquad
    K_{(1,1,1),(1^3)}(q)=1.
  \]
  Since $(3)^T=(1,1,1)$ and $(2,1)^T=(2,1)$, the main theorem yields
  \begin{align*}
    T_{X,X^!}(x,y)
      &=x^3+(x+x^2)(y+y^2)+y^3\\
      &=x^3+y^3+xy+x^2y+xy^2+x^2y^2.
  \end{align*}
  In particular, $T_{X,X^!}(1,1)=6=|S_3|$.
\end{example}
